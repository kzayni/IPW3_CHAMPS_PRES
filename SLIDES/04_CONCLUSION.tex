\section[Conclusion]{\textbf{Conclusion}}

% ============================================================
% CONCLUSIONS
% ============================================================

\begin{frame}{\small Conclusions}
\scriptsize

% ------------------------------------------------------------
% NACA0012
% ------------------------------------------------------------
\begin{block}{\scriptsize NACA0012}
\begin{itemize}
    \item CHAMPS predictions are generally well positioned within the IPW3 code-to-code envelope, with limited grid sensitivity.
    \item Eulerian and Lagrangian droplet formulations show good agreement in the predicted collection-efficiency distributions.
    \item Single-layer ice shapes capture the experimental horn orientation well, while multilayer simulations reproduce complex three-dimensional features.
\end{itemize}
\end{block}

\vspace{0.15cm}

% ------------------------------------------------------------
% ONERA M6
% ------------------------------------------------------------
\begin{block}{\scriptsize ONERA M6}
\begin{itemize}
    \item CHAMPS predictions remain consistent with the overall IPW3 trends across the different physical quantities.
    \item Single-layer ice shapes remain consistent across the grid hierarchy, with L4 exhibiting the largest deviations due to its coarser surface discretization.
    \item Multilayer simulations capture a more pronounced horn formation compared with the single-layer prediction.
\end{itemize}
\end{block}

\vspace{0.15cm}

% ------------------------------------------------------------
% OVERALL
% ------------------------------------------------------------
\begin{alertblock}{\scriptsize Overall}
\begin{itemize}
    \item CHAMPS demonstrates a consistent end-to-end icing capability, with the level-set framework capturing complex three-dimensional ice morphologies in multilayer simulations.
    \item Future work will focus on improving the roughness and ice-density models.
\end{itemize}
\end{alertblock}

\end{frame}
